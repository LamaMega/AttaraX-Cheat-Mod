Timer.schedule(() => {

    Groups.build.each(build => {

        // ONLY yellow team
        if(build.team != Team.sharded) return;
        //and
        if(build.block instanceof CoreBlock) return;

        // Get block consumers
        let consumers = build.block.consumers;

        if(consumers == null) return;

        consumers.each(cons => {

            // Item consumers
            if(cons instanceof ConsumeItems){

                cons.items.each(stack => {

                    // Fill only needed item
                    build.items.set(stack.item, 1);
                });
            }});
